# /atari.ini - the classic Atari's settings, read by the I/O processor at power-on and
# at a full restart. Settings > Classic Atari edits it (only the lines of these keys).
# ram: 128 or 320 (KB; 320 = RAMBO)
ram=320
# kbdlayout: 0 = Atari key positions, 1 = US PC symbols
kbdlayout=0
# joystick: 1 = the arrow keys are joystick 1 (Left Alt = fire)
joystick=0
# scanlines: 0 off, 1 = 25%, 2 = 50%, 3 = 75%
scanlines=2
# hpos: horizontal position of the picture, 0-48
hpos=0
# boot=ash: /ASH.CAR is the default cartridge (else BASIC)
boot=basic
